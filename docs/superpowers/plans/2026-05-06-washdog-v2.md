```kotlin
package com.washdog.app.ui.components

import androidx.compose.foundation.text.KeyboardOptions
import androidx.compose.material3.OutlinedTextField
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.AnnotatedString
import androidx.compose.ui.text.input.KeyboardType
import androidx.compose.ui.text.input.OffsetMapping
import androidx.compose.ui.text.input.TransformedText
import androidx.compose.ui.text.input.VisualTransformation

private const val PHONE_PREFIX = "+52 "

private object PhoneMaskTransformation : VisualTransformation {
    override fun filter(text: AnnotatedString): TransformedText {
        val out = AnnotatedString(PHONE_PREFIX + text.text)
        val mapping = object : OffsetMapping {
            override fun originalToTransformed(offset: Int) = offset + PHONE_PREFIX.length
            override fun transformedToOriginal(offset: Int) =
                (offset - PHONE_PREFIX.length).coerceAtLeast(0).coerceAtMost(text.length)
        }
        return TransformedText(out, mapping)
    }
}

@Composable
fun WDPhoneField(
    digits: String,
    onDigitsChange: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    OutlinedTextField(
        value                = digits,
        onValueChange        = { new ->
            val filtered = new.filter { it.isDigit() }.take(10)
            onDigitsChange(filtered)
        },
        label                = { Text("Teléfono") },
        placeholder          = { Text("10 dígitos") },
        visualTransformation = PhoneMaskTransformation,
        keyboardOptions      = KeyboardOptions(keyboardType = KeyboardType.Number),
        singleLine           = true,
        modifier             = modifier
    )
}

fun digitsToPhone(digits: String): String = "+52$digits"

fun phoneToDigits(phone: String): String =
    phone.removePrefix("+52").removePrefix(" ").filter { it.isDigit() }
```

```kotlin
package com.washdog.app.util

import com.washdog.app.data.models.Appointment

object OverlapChecker {

    fun findConflict(candidate: Appointment, existing: List<Appointment>): Appointment? {
        val candStart = toMinutes(candidate.time) ?: return null
        val candEnd   = toMinutes(candidate.endTime)
            .takeIf { candidate.endTime.isNotEmpty() } ?: (candStart + 60)

        return existing.firstOrNull { appt ->
            if (appt.id == candidate.id)      return@firstOrNull false
            if (appt.date != candidate.date)  return@firstOrNull false
            val exStart = toMinutes(appt.time) ?: return@firstOrNull false
            val exEnd   = toMinutes(appt.endTime)
                .takeIf { appt.endTime.isNotEmpty() } ?: (exStart + 60)
            candStart < exEnd && candEnd > exStart
        }
    }

    internal fun toMinutes(time: String): Int? {
        if (time.isBlank()) return null
        val parts = time.split(":").mapNotNull { it.toIntOrNull() }
        return if (parts.size == 2) parts[0] * 60 + parts[1] else null
    }
}
```

```kotlin
                            .padding(start = 4.dp, top = 2.dp)
                    )
                    Box(
                        Modifier
                            .weight(1f)
                            .fillMaxHeight()
                            .background(Color(0x08000000))
                    )
                }
            }
        }

        // ── Bloques de citas ─────────────────────────────────────────────────
        appointments.forEach { appt ->
            val startMin = toMin(appt.time) ?: return@forEach
            if (startMin < FIRST_H * 60 || startMin >= LAST_H * 60) return@forEach

            val endMin   = toMin(appt.endTime)
                .takeIf { appt.endTime.isNotEmpty() } ?: (startMin + 60)
            val topDp    = SLOT_H * ((startMin - FIRST_H * 60).toFloat() / 60f)
            val heightDp = (SLOT_H * ((endMin - startMin).toFloat() / 60f))
                .coerceAtLeast(44.dp)

            Box(
                modifier = Modifier
                    .padding(start = LABEL_W + 4.dp, end = 8.dp)
                    .absoluteOffset(y = topDp)
                    .fillMaxWidth()
                    .height(heightDp)
                    .clip(RoundedCornerShape(8.dp))
                    .background(appt.status.blockColor())
                    .padding(horizontal = 8.dp, vertical = 4.dp)
            ) {
                Column {
                    Text(
                        text  = if (appt.endTime.isNotEmpty()) "${appt.time}–${appt.endTime}"
                                else appt.time,
                        fontSize = 10.sp,
                        color    = Color.White.copy(alpha = 0.85f)
                    )
                    Text(
                        text       = appt.clientName,
                        fontSize   = 12.sp,
                        fontWeight = FontWeight.SemiBold,
                        color      = Color.White
                    )
                    Text(
                        text     = appt.service,
                        fontSize = 10.sp,
                        color    = Color.White.copy(alpha = 0.8f),
                        maxLines = 1
                    )
                }
            }
        }
    }
}

private fun toMin(time: String): Int? {
    if (time.isBlank()) return null
    val parts = time.split(":").mapNotNull { it.toIntOrNull() }
    return if (parts.size == 2) parts[0] * 60 + parts[1] else null
}

private fun ApptStatus.blockColor(): Color = when (this) {
    ApptStatus.DONE      -> Color(0xFF9E9E9E)
    ApptStatus.NEXT      -> WDPrimary
    ApptStatus.SCHEDULED -> Color(0xFF5B8C7E)
    ApptStatus.REMINDER  -> WDWarn
}
```

```kotlin
package com.washdog.app

import com.washdog.app.data.models.Appointment
import com.washdog.app.data.models.ApptStatus
import com.washdog.app.util.OverlapChecker
import org.junit.Assert.assertNotNull
import org.junit.Assert.assertNull
import org.junit.Test

class OverlapDetectionTest {

    private fun appt(id: String, date: String, start: String, end: String = "") = Appointment(
        id         = id,
        clientName = "Cliente $id",
        petName    = "Mascota $id",
        service    = "Baño",
        time       = start,
        endTime    = end,
        date       = date,
        status     = ApptStatus.SCHEDULED
    )

    private val DATE = "2026-05-10"

    @Test fun `no overlap when adjacent — end of A equals start of B`() {
        val existing = listOf(appt("1", DATE, "09:00", "10:00"))
        val candidate = appt("2", DATE, "10:00", "11:00")
        assertNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `overlap when ranges intersect`() {
        val existing = listOf(appt("1", DATE, "09:00", "11:00"))
        val candidate = appt("2", DATE, "10:00", "12:00")
        assertNotNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `overlap when candidate is fully contained`() {
        val existing = listOf(appt("1", DATE, "09:00", "12:00"))
        val candidate = appt("2", DATE, "10:00", "11:00")
        assertNotNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `no overlap when different dates`() {
        val existing = listOf(appt("1", "2026-05-10", "09:00", "11:00"))
        val candidate = appt("2", "2026-05-11", "09:00", "11:00")
        assertNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `no overlap with same id — updating same appointment`() {
        val existing = listOf(appt("1", DATE, "09:00", "10:00"))
        val candidate = appt("1", DATE, "09:00", "10:00") // same id
        assertNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `overlap defaults to 1h when endTime is empty`() {
        // existing: 09:00 (no end → assumed until 10:00)
        val existing = listOf(appt("1", DATE, "09:00"))
        // candidate: 09:30 (no end → assumed until 10:30) → overlaps
        val candidate = appt("2", DATE, "09:30")
        assertNotNull(OverlapChecker.findConflict(candidate, existing))
    }

    @Test fun `no overlap when candidate starts after default-1h window`() {
        val existing = listOf(appt("1", DATE, "09:00"))  // ends at 10:00
        val candidate = appt("2", DATE, "10:00")         // starts at 10:00 — adjacent
        assertNull(OverlapChecker.findConflict(candidate, existing))
    }
}
```
